# SA-Live-Transcribe V0, pinned dependencies
#
# Tested with Python 3.12 on Windows 11. ctranslate2 wheels do not yet ship
# for Python 3.13/3.14, use `py -3.12 -m venv .venv`.
#
# ctranslate2 4.7.x bundles cuDNN 9 directly in the wheel, no separate
# cuDNN install needed. CUDA 12 runtime/driver is still required for GPU.
# CPU-only users: just `pip install -r requirements.txt`.

faster-whisper==1.2.1
ctranslate2==4.7.2

# Audio capture (Windows): PyAudio fork with WASAPI loopback support.
# `sounddevice` doesn't expose loopback; `soundcard` advertises it but returns
# silence on many Windows audio configurations (confirmed on Sean's Realtek
# setup). Windows-only: the macOS backend uses sounddevice + the audiotap helper.
PyAudioWPatch==0.2.12.8; sys_platform == "win32"

# Audio capture (macOS): mic via sounddevice (PortAudio over Core Audio,
# universal2 wheel). System audio is captured by the signed volksmond-audiotap
# helper (a Core Audio process tap), not by sounddevice. darwin-only.
sounddevice==0.5.1; sys_platform == "darwin"

# Audio capture (Linux): native libpulse, no PortAudio. Released PortAudio
# builds have no PulseAudio/PipeWire host API, so monitor sources (the Linux
# loopback equivalent) are invisible to sounddevice there. pulsectl (ctypes
# over libpulse.so.0) enumerates sources; the default sink's ".monitor" source
# is the system-audio tap. pasimple (libpulse simple API, also ctypes-only)
# records both the MIC and SYS streams. One client stack covers PulseAudio
# (Mint 21.x) and pipewire-pulse (Mint 22, Debian 12+) alike. linux-only.
pulsectl==24.12.0; sys_platform == "linux"
pasimple==0.0.3; sys_platform == "linux"

# 48k/44.1k -> 16k resampling (polyphase filter, alias-safe)
scipy==1.13.1
numpy==1.26.4

# Echo cancellation (optional, opt-in). livekit exposes Chrome's WebRTC APM (AEC3) for both the
# re-transcribe canceller (live_transcribe/aec.py) and the live one (live_transcribe/aec_live.py);
# soxr does the streaming resample for the live path. Both imported lazily, so the app runs
# without them (echo cancellation just stays off). Native libs; bundled in the packaged build.
livekit==1.1.11
soxr==1.1.0

# Tier auto-detect reads total RAM as a fallback when no CUDA device is found
psutil==5.9.8

# Browser UI (FastAPI + uvicorn). Optional, only needed if you use the web UI.
fastapi==0.136.1
uvicorn[standard]==0.47.0

# Note: torch is NOT required. CUDA presence is detected via
# ctranslate2.get_cuda_device_count(), already a transitive dep.

# Optional: Outlook calendar attendee-seeding (--seed-from-calendar). MSAL device-
# code auth + Graph REST. requests comes in transitively but is pinned for repro.
# Imported lazily, so the tool runs without these unless the flag is used.
msal==1.36.0
requests==2.34.2

# Optional: LOCAL Outlook calendar attendee-seeding (live_transcribe/outlook_local.py, the
# offline alternative to the Graph path above). Reads the classic Outlook desktop app over COM,
# no network call. Windows only, imported lazily so the app runs without it (the in-app
# "Pull from Outlook calendar" button just reports it is unavailable).
pywin32==311; sys_platform == "win32"

# Offline licence signing/verification (live_transcribe/licensing.py): Ed25519 via
# cryptography. Already present transitively via msal, pinned here for repro.
cryptography==48.0.0

# Optional: native desktop-app shell (python -m live_transcribe.desktop), the
# basis for the packaged installer. Browser UI works without it. Pulls pythonnet.
pywebview==6.2.1

# Optional: local AI summaries (live_transcribe/summarise.py). Runs a GGUF model
# in-process via llama.cpp, no daemon. On Windows install the prebuilt CPU wheel:
#   pip install llama-cpp-python==0.3.22 --extra-index-url https://abetlen.github.io/llama-cpp-python/whl/cpu
# Imported lazily, so the tool runs without it unless summaries are used.
# PINNED to 0.3.22: its cpu wheel is AVX2-safe (0 AVX-512 instructions), so it runs on AVX2-only
# CPUs (e.g. i7-9750H). The 0.3.23 cpu wheel is compiled WITH AVX-512 and crashes there with
# STATUS_ILLEGAL_INSTRUCTION. The build guard (tools/ensure_avx2_llama.py) re-verifies by
# disassembly. Both versions still load Gemma 4.
#
# GPU SUMMARIES (optional): the CPU wheel above runs summaries on the CPU. To run them on an
# NVIDIA GPU, install the matching CUDA wheel INSTEAD (abetlen ships prebuilt CUDA wheels):
#   pip install llama-cpp-python==0.3.22 --extra-index-url https://abetlen.github.io/llama-cpp-python/whl/cu124
# (use the cu12x that matches your CUDA 12 runtime). The app detects this at runtime via
# llama_cpp.llama_supports_gpu_offload() and offloads the summary model to the GPU when it
# fits in VRAM; with the CPU wheel that returns False and summaries stay on the CPU. Shipping
# this in the packaged app belongs in the separate GPU build (see docs/cuda-build-plan.md).
llama-cpp-python==0.3.22
